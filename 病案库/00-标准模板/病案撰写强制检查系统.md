## 三、撰写后强制验证（写后必过）

### 3.1 自动验证脚本

**在命令行运行以下验证**（必须全部通过）：

```bash
#!/bin/bash
# 病案撰写验证脚本 V1.0
# 使用方法：./verify_case.sh 文件名.md

FILE="$1"
ERRORS=0

echo "========================================"
echo "病案撰写强制验证"
echo "文件: $FILE"
echo "========================================"
echo ""

# 1. 检查模块数量（必须≥8）
MODULE_COUNT=$(grep -c "^## 模块" "$FILE")
if [ "$MODULE_COUNT" -lt 8 ]; then
    echo "❌ 模块数量不足: $MODULE_COUNT (需要≥8)"
    ERRORS=$((ERRORS + 1))
else
    echo "✅ 模块数量: $MODULE_COUNT"
fi

# 2. 检查模块一至八是否齐全
for i in 一 二 三 四 五 六 七 八; do
    if ! grep -q "## 模块$i" "$FILE"; then
        echo "❌ 缺失模块$i"
        ERRORS=$((ERRORS + 1))
    fi
done
if [ $? -eq 0 ]; then
    echo "✅ 模块一至八齐全"
fi

# 3. 检查十八变全程预测（模块九）
if ! grep -q "十八变全程预测\|## 模块九" "$FILE"; then
    echo "❌ 缺失十八变全程预测（模块九）"
    ERRORS=$((ERRORS + 1))
else
    echo "✅ 十八变全程预测已包含"
fi

# 4. 检查通神露泡脚方案（模块十）
if ! grep -q "通神露泡脚方案\|## 模块十" "$FILE"; then
    echo "❌ 缺失通神露泡脚方案（模块十）"
    ERRORS=$((ERRORS + 1))
else
    echo "✅ 通神露泡脚方案已包含"
fi

# 5. 检查食疗决策公式
if ! grep -q "今日食疗 = f(" "$FILE"; then
    echo "❌ 缺失食疗决策公式"
    ERRORS=$((ERRORS + 1))
else
    echo "✅ 食疗决策公式已包含"
fi

# 6. 检查音乐疗法决策公式
if ! grep -q "今日音乐 = f(\|五音疗法.*决策公式" "$FILE"; then
    echo "❌ 缺失音乐疗法决策公式"
    ERRORS=$((ERRORS + 1))
else
    echo "✅ 音乐疗法决策公式已包含"
fi

# 7. 检查发酵食物强制方案
if ! grep -q "发酵食物不是建议，是必须" "$FILE"; then
    echo "❌ 缺失发酵食物核心警告"
    ERRORS=$((ERRORS + 1))
else
    echo "✅ 发酵食物核心警告已包含"
fi

# 8. 检查五维诊断完整性
WUXI_DIMENSIONS=0
for dim in 炁诊断 毒诊断 脉诊断 邪诊断 瘾诊断; do
    if grep -q "$dim" "$FILE"; then
        WUXI_DIMENSIONS=$((WUXI_DIMENSIONS + 1))
    fi
done
if [ "$WUXI_DIMENSIONS" -lt 5 ]; then
    echo "❌ 五维诊断不完整: $WUXI_DIMENSIONS/5"
    ERRORS=$((ERRORS + 1))
else
    echo "✅ 五维诊断完整: 5/5"
fi

# 9. 检查时辰位移诊断
if ! grep -q "时辰位移" "$FILE"; then
    echo "❌ 缺失时辰位移诊断"
    ERRORS=$((ERRORS + 1))
else
    echo "✅ 时辰位移诊断已包含"
fi

# 10. 检查54张牌治疗方案
if ! grep -q "54张牌\|54牌" "$FILE"; then
    echo "❌ 缺失54张牌治疗方案"
    ERRORS=$((ERRORS + 1))
else
    echo "✅ 54张牌治疗方案已包含"
fi

echo ""
echo "========================================"
if [ $ERRORS -eq 0 ]; then
    echo "✅ 验证通过！可以提交。"
    exit 0
else
    echo "❌ 验证失败！发现 $ERRORS 个错误。"
    echo "请先修正错误后再提交。"
    exit 1
fi
```
